// ── State ──

let cfg = { time: 20, reps: 10, users: 1, names: ['User 1'] };

let state = { currentRep: 0, currentUserIdx: 0, remaining: 20, interval: null, running: false };


const USER_COLORS = ['#f0c040','#40c8f0','#e05a30','#a070e0','#40c878','#e050a0'];


// ── Audio ──

let audioCtx = null;


function ensureAudio() {

  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

  if (audioCtx.state === 'suspended') audioCtx.resume();

}


function playBeep() {

  try {

    ensureAudio();

    // Three stacked oscillators for a loud, piercing beep

    const layers = [

      { type: 'square', freq: 880,  gain: 0.5 },

      { type: 'square', freq: 1320, gain: 0.35 },

      { type: 'sine',   freq: 660,  gain: 0.3  },

    ];

    const masterGain = audioCtx.createGain();

    masterGain.gain.setValueAtTime(1, audioCtx.currentTime);

    masterGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1.2);

    masterGain.connect(audioCtx.destination);


    layers.forEach(({ type, freq, gain }) => {

      const osc = audioCtx.createOscillator();

      const g = audioCtx.createGain();

      osc.type = type;

      osc.frequency.value = freq;

      g.gain.value = gain;

      osc.connect(g);

      g.connect(masterGain);

      osc.start(audioCtx.currentTime);

      osc.stop(audioCtx.currentTime + 1.2);

    });

  } catch(e) { console.warn('Audio error:', e); }

}


// ── Setup ──

let setupTime = 20, setupReps = 10, setupUsers = 1;


function adjustTime(d) {

  setupTime = Math.max(1, Math.min(300, setupTime + d));

  document.getElementById('time-display').value = setupTime;

}

function syncTime(el) {

  const v = parseInt(el.value);

  if (!isNaN(v) && el.value !== '') setupTime = Math.max(1, Math.min(300, v));

}


function adjustReps(d) {

  setupReps = Math.max(1, Math.min(99, setupReps + d));

  document.getElementById('reps-display').value = setupReps;

}

function syncReps(el) {

  const v = parseInt(el.value);

  if (!isNaN(v) && el.value !== '') setupReps = Math.max(1, Math.min(99, v));

}


function adjustUsers(d) {

  setupUsers = Math.max(1, Math.min(6, setupUsers + d));

  document.getElementById('users-display').value = setupUsers;

  renderUserInputs();

}

function syncUsers(el) {

  const v = parseInt(el.value);

  if (!isNaN(v) && el.value !== '' && v >= 1 && v <= 6) {

    setupUsers = v;

    renderUserInputs();

  }

}


function renderUserInputs() {

  const container = document.getElementById('user-inputs');

  const existing = container.querySelectorAll('.user-name-input');

  const vals = Array.from(existing).map(i => i.value);

  container.innerHTML = '';

  for (let i = 0; i < setupUsers; i++) {

    const color = USER_COLORS[i % USER_COLORS.length];

    const row = document.createElement('div');

    row.className = 'user-row';

    row.innerHTML = `

      <div class="user-dot" style="background:${color}"></div>

      <input class="user-name-input" type="text" maxlength="20"

        placeholder="User ${i + 1}"

        value="${vals[i] || ''}" />

    `;

    container.appendChild(row);

  }

}


function startWorkout() {

  ensureAudio(); // unlock audio on user gesture

  const inputs = document.querySelectorAll('.user-name-input');

  const names = Array.from(inputs).map((inp, i) => inp.value.trim() || `User ${i + 1}`);

  cfg = { time: setupTime, reps: setupReps, users: setupUsers, names };


  // userRepCount[i] = how many reps user i has completed

  const userRepCount = new Array(cfg.users).fill(0);

  state = { currentTurn: 0, remaining: cfg.time, interval: null, running: false, userRepCount };


  document.getElementById('setup-screen').style.display = 'none';

  document.getElementById('app-screen').style.display = 'flex';

  document.getElementById('done-overlay').style.display = 'none';


  renderQueue();

  updateTimerPanel();

}


// Total turns = cfg.users * cfg.reps

function totalTurns() { return cfg.users * cfg.reps; }


// For a given turn index, which user and which rep# (1-based) is it?

function turnInfo(turn) {

  const userIdx = turn % cfg.users;

  const repNum = Math.floor(turn / cfg.users) + 1; // 1-based rep for this user

  return { userIdx, repNum };

}


// ── Queue Panel (Panel 1) ──

function renderQueue() {

  const queueRow = document.getElementById('queue-row');

  const dotsEl = document.getElementById('progress-dots');

  queueRow.innerHTML = '';

  dotsEl.innerHTML = '';


  const total = totalTurns();

  // Show current + up to 3 upcoming

  const lookahead = Math.min(4, total - state.currentTurn);

  for (let i = 0; i < lookahead; i++) {

    const t = state.currentTurn + i;

    if (t >= total) break;

    const { userIdx, repNum } = turnInfo(t);

    const color = USER_COLORS[userIdx % USER_COLORS.length];

    const card = document.createElement('div');

    card.className = 'queue-card ' + (i === 0 ? 'current' : 'upcoming');

    card.innerHTML = `

      <div class="queue-card-label">${i === 0 ? '▶ NOW' : 'NEXT ' + (i)}</div>

      <div class="queue-card-name" style="color:${i === 0 ? color : ''}">${cfg.names[userIdx]}</div>

      <div class="queue-card-rep" style="color:${i === 0 ? color : ''}">Rep ${repNum} of ${cfg.reps}</div>

    `;

    queueRow.appendChild(card);

  }


  // Progress dots — one per total turn

  const maxDots = Math.min(total, 40); // cap at 40 dots

  for (let t = 0; t < maxDots; t++) {

    const dot = document.createElement('div');

    dot.className = 'dot' + (t < state.currentTurn ? ' done' : t === state.currentTurn ? ' active' : '');

    dotsEl.appendChild(dot);

  }

}


// ── Timer Panel (Panel 2) ──

function updateTimerPanel() {

  const { userIdx, repNum } = turnInfo(state.currentTurn);

  const color = USER_COLORS[userIdx % USER_COLORS.length];

  const nameEl = document.getElementById('current-user-name');

  nameEl.textContent = cfg.names[userIdx];

  nameEl.style.color = color;


  document.getElementById('rep-counter-label').textContent = `REP ${repNum} OF ${cfg.reps}`;


  setTimerDisplay(state.remaining);

  setProgress(state.remaining, cfg.time);


  const goBtn = document.getElementById('go-btn');

  const label = document.getElementById('current-user-label');

  goBtn.textContent = state.running ? 'RUNNING…' : 'START';

  goBtn.className = state.running ? 'running' : '';

  label.textContent = state.running ? 'NOW EXERCISING' : 'UP NEXT';

}


function setTimerDisplay(secs) {

  const el = document.getElementById('timer-display');

  el.textContent = secs;

  el.className = secs <= 3 ? 'danger' : secs <= Math.ceil(cfg.time * 0.4) ? 'warning' : '';

}


function setProgress(remaining, total) {

  const pct = (remaining / total) * 100;

  const bar = document.getElementById('progress-bar');

  bar.style.width = pct + '%';

  bar.style.background = remaining <= 3 ? 'var(--red)' : remaining <= Math.ceil(total * 0.4) ? 'var(--accent2)' : 'var(--accent)';

}


// ── Timer Logic ──

function handleGoBtn() {

  if (state.running) return;

  startCountdown();

}


function startCountdown() {

  state.remaining = cfg.time;

  state.running = true;

  updateTimerPanel(); // updates label to NOW EXERCISING


  state.interval = setInterval(() => {

    state.remaining--;

    setTimerDisplay(state.remaining);

    setProgress(state.remaining, cfg.time);


    if (state.remaining <= 0) {

      clearInterval(state.interval);

      state.running = false;

      repComplete();

    }

  }, 1000);

}


function repComplete() {

  // Loud beep

  playBeep();


  // Flash done

  const el = document.getElementById('timer-display');

  el.className = 'done-flash';

  el.textContent = '✓';


  // Vibrate if supported

  if (navigator.vibrate) navigator.vibrate([150, 80, 150, 80, 150]);


  setTimeout(() => {

    state.currentTurn++;

    if (state.currentTurn >= totalTurns()) {

      showWorkoutDone();

    } else {

      state.remaining = cfg.time;

      renderQueue();

      updateTimerPanel();

    }

  }, 800);

}


function showWorkoutDone() {

  document.getElementById('done-overlay').style.display = 'flex';

  const totalTime = totalTurns() * cfg.time;

  const mins = Math.floor(totalTime / 60);

  const secs = totalTime % 60;

  document.getElementById('done-summary').textContent =

    `${cfg.reps} REPS · ${cfg.users} USER${cfg.users > 1 ? 'S' : ''} · ${mins > 0 ? mins + 'm ' : ''}${secs}s`;

}


function goSetup() {

  clearInterval(state.interval);

  document.getElementById('app-screen').style.display = 'none';

  document.getElementById('setup-screen').style.display = 'flex';

  renderUserInputs();

}


// Init

renderUserInputs();
